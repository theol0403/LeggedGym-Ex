% Script:

\begin{frame}{What Is Quadruped Parkour?}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
    \begin{block}{Working Definition}
      \small
      Quadruped parkour is \alert{dynamic locomotion over discrete obstacles}
      such as stairs, hurdles, and gaps, where success depends on
      \begin{itemize}
        \item perceiving terrain geometry,
        \item choosing the right footholds and timing,
        \item staying stable through contact changes.
      \end{itemize}
    \end{block}

    \column{0.26\textwidth}
    \centering
    \includegraphics[width=\linewidth]{parkour_obstacle_types.png}

  \end{columns}

\end{frame}

% Script:
% - This slide now carries only the motivation and thesis question.
% - Emphasize that I am not claiming a solved deployment pipeline; this is an experimental study.
% - The main question is whether pretrained monocular depth can preserve useful geometry.
\begin{frame}{Motivation and Thesis Question}
  \begin{columns}[T,onlytextwidth]
    \column{0.56\textwidth}
    \begin{itemize}
      \item Most visual parkour systems deploy with an onboard \alert{depth camera}.
      \item RGB is easier to deploy, but the visual sim-to-real gap is harder.
      \item I studied whether \alert{monocular depth} can act as an intermediate representation.
    \end{itemize}

    \vspace{0.5em}
    \begin{alertblock}{Thesis Question}
      Can a pretrained monocular depth estimator preserve enough terrain geometry for a distilled quadruped parkour policy to act reliably?
    \end{alertblock}

    \column{0.38\textwidth}
    \begin{block}{Contribution Style}
      \small
      This thesis is mainly an \alert{experimental and methodological study}:
      what I tried, what worked, what failed, and what that says about the role of perception.
    \end{block}
  \end{columns}
\end{frame}

\section{Method}
% ==========================================================================

\sectionslide{QueensRed}{Method}

% Script:
% - Keep the formal setup, but make it lighter.
% - The main modeling idea is partial observability with different exteroceptive channels for teacher and student.
\begin{frame}{Mathematical Model of the Task}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{POMDP}
      \small
      \[
        (\mathcal{S}, \mathcal{A}, \mathcal{O}, T, R, \gamma),
        \qquad
        J(\pi)=\E_{\pi}\!\left[\sum_{t=0}^{\infty}\gamma^t R(s_t,a_t)\right]
      \]
      with actions \(a_t \in \R^{12}\).
    \end{block}

    \begin{block}{Observation Split}
      \small
      \[
        o_t=(o_{\mathrm{prop}},o_{\mathrm{ext}})
      \]
      Proprioception carries robot state; exteroception carries terrain geometry.
    \end{block}

    \column{0.44\textwidth}
    \begin{block}{Teacher}
      \small
      \[
        o_t^T=(o_{\mathrm{prop}},\text{scandots},\text{priv.\ params})
      \]
      Direct terrain geometry during training.
    \end{block}

    \begin{block}{Student}
      \small
      \[
        o_t^S=(o_{\mathrm{prop}},\text{depth image})
      \]
      Partial observation, so a recurrent encoder is justified.
    \end{block}
  \end{columns}
\end{frame}

% Script:
% - This is the method slide that should stay visual.
% - I want the audience to remember the separation between control learning and perception transfer.
\begin{frame}{Two-Stage Pipeline}
  \begin{columns}[T,onlytextwidth]
    \column{0.56\textwidth}
    \centering
    \scalebox{0.80}{\input{../thesis/figures/pipeline}}

    \column{0.38\textwidth}
    \begin{itemize}
      \item \textbf{Teacher:} PPO with privileged geometry.
      \item \textbf{Student:} DAgger from the student's own state distribution.
      \item \textbf{Question:} replace simulator depth with DA2-estimated monocular depth.
    \end{itemize}

    \vspace{0.4em}
    \begin{alertblock}{Main idea}
      First solve control, then test whether perception preserves the geometry the controller needs.
    \end{alertblock}
  \end{columns}
  \source{Teacher--student pipeline adapted from the thesis; PPO after Schulman et al.\ (2017), DAgger after Ross et al.\ (2011).}
\end{frame}

% Script:
% - DSQ is the compact mathematical diagnostic of the thesis.
% - The key reason for using correlation is monocular scale-shift ambiguity.
\begin{frame}{Depth Signal Quality (DSQ)}
  \begin{columns}[T,onlytextwidth]
    \column{0.56\textwidth}
    \begin{block}{Definition}
      \small
      \[
        \DSQ
        =
        \frac{1}{T}\sum_{t=1}^{T}
        \frac{1}{N}\sum_{i=1}^{N}
        \rho\!\left(d_i^{\mathrm{GT}},d_i^{\mathrm{DA2}}\right)
      \]
      where \(\rho\) is Pearson correlation over pixels.
    \end{block}

    \begin{alertblock}{Why this metric?}
      \small
      Pearson correlation is invariant to positive affine transforms, so DSQ ignores the scale--shift ambiguity of monocular depth and focuses on \alert{preserved geometry}.
    \end{alertblock}

    \column{0.38\textwidth}
    \visualplaceholder{DSQ explainer visual}{
      Suggested visual: side-by-side depth frames labeled ``same shape, different scale'' versus ``geometry collapsed''.}
  \end{columns}
\end{frame}

\begin{frame}{Main Findings}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
    \begin{block}{Baseline message}
      \small
      Teacher and GT-depth student are strong; zero-depth almost fails completely.
    \end{block}

    \begin{block}{Estimated-depth message}
      \small
      Larger DA2 models help, and texture helps the hardest obstacle class: \alert{gaps}.
    \end{block}

    \begin{alertblock}{Takeaway}
      \small
      The main bottleneck is \alert{perception quality}, not the downstream controller.
    \end{alertblock}

  \end{columns}
\end{frame}

\begin{frame}{Sensitivity and the Role of DSQ}
  \begin{columns}[T,onlytextwidth]
    \column{0.54\textwidth}
    \centering
    \includegraphics[width=0.96\linewidth]{dsq_scatter.pdf}

    \column{0.40\textwidth}
    \begin{block}{Observed pattern}
      \small
      \begin{itemize}
        \item Lighting and color changes are often tolerable.
        \item Featureless surfaces cause the largest failures.
        \item Low DSQ usually coincides with failure.
      \end{itemize}
    \end{block}

    \begin{alertblock}{Interpretation}
      \small
      DSQ is a useful \alert{diagnostic}, but high DSQ alone does not guarantee successful control.
    \end{alertblock}

\section{Conclusion}
% ==========================================================================

% Script:
% - End with a short and honest synthesis.
% - This should sound like an exploration that clarified the right question, not a claim of full deployment readiness.
\begin{frame}{Takeaways and Limits}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{What I learned}
      \small
      \begin{itemize}
        \item Monocular depth can support nontrivial parkour \alert{in simulation}.
        \item Teacher--student training is a useful way to isolate perception from control.
        \item Geometry preservation matters, and DSQ captures part of that story.
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{What remains open}
      \small
      \begin{itemize}
        \item Real-world transfer is untested.
        \item Texture-poor surfaces are still a major weakness.
        \item The current depth stack is too heavy for onboard deployment.
      \end{itemize}
    \end{block}
  \end{columns}

  \vspace{0.4em}
  \begin{alertblock}{Bottom line}
    This thesis suggests that monocular depth is a plausible experimental direction for parkour, but perception quality remains the dominant bottleneck.
  \end{alertblock}
\end{frame}
