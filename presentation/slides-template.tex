% Section divider: colored background + centered white title
% Usage: \sectionslide{QueensRed}{Title Text}
\newcommand{\sectionslide}[2]{%
  \begingroup
  \setbeamercolor{background canvas}{bg=#1}
  \begin{frame}[plain,c]
    \centering{\Huge\bfseries\color{white}#2}
  \end{frame}
  \endgroup
}
